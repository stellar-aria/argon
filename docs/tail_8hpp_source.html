78template <size_t Stride, typename ScalarType>
79typename Partial<ScalarType, Stride>::vector_type load_partial(const ScalarType* ptr, size_t count) {
80 using argon_type = Argon<ScalarType>;
81 using predicate_type = typename argon_type::argon_bool_type;
82 if constexpr (Stride == 1) {
83#ifdef ARGON_PLATFORM_MVE
84 // Always predicate: a vctp-predicated loop is what the compiler turns into dlstp/letp.
85 return argon_type::Load(ptr, predicate_type::FirstN(count));
86#else
87 return count >= argon_type::lanes ? argon_type::Load(ptr) : argon_type::Load(ptr, predicate_type::FirstN(count));
88#endif
89 } else {
90 if (count >= argon_type::lanes) {
91 return argon_type::template LoadInterleaved<Stride>(ptr);
92 }
93 // De-interleaving loads (vld2/vld4) can't be predicated, even on MVE: gather each channel instead.
94 const auto offsets = frame_offsets<ScalarType, Stride>();
95 const auto active = predicate_type::FirstN(count);
96 std::array<argon_type, Stride> out;
97 for (size_t channel = 0; channel < Stride; ++channel) {
98 out[channel] = argon_type::LoadGatherOffsetIndex(ptr + channel, offsets, active);
99 }
100 return out;
101 }
102}

105template <size_t Stride, typename ScalarType>
106void store_partial(ScalarType* ptr, const typename Partial<ScalarType, Stride>::vector_type& value, size_t count) {
107 using argon_type = Argon<ScalarType>;
108 using predicate_type = typename argon_type::argon_bool_type;
109 if constexpr (Stride == 1) {
110#ifdef ARGON_PLATFORM_MVE
111 value.StoreTo(ptr, predicate_type::FirstN(count));
112#else
113 if (count >= argon_type::lanes) {
114 value.StoreTo(ptr);
115 } else {
116 value.StoreTo(ptr, predicate_type::FirstN(count));
117 }
118#endif
119 } else {
120 if (count >= argon_type::lanes) {
121 argon::store_interleaved(ptr, value);
122 return;
123 }
124 // Interleaving stores (vst2/vst4) can't be predicated, even on MVE: scatter each channel instead.
125 const auto offsets = frame_offsets<ScalarType, Stride>();
126 const auto active = predicate_type::FirstN(count);
127 for (size_t channel = 0; channel < Stride; ++channel) {
128 value[channel].StoreScatterOffsetIndex(ptr + channel, offsets, active);
129 }
130 }
