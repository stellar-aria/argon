623 ace argon_type ReciprocalSqrtEstimate() const
624 requires lane_floating_point<scalar_type> || (std::is_same_v<scalar_type, uint32_t> && !mve_platform)
625 {
626#ifdef ARGON_PLATFORM_MVE
627 // MVE has no vrsqrte: take the bit-level initial guess, then one Newton-Raphson step, which lands within NEON's
628 // estimate precision (~1/256).
629 const auto bits = std::bit_cast<Bool_t<VectorType>>(vec_);
630 const VectorType guess = std::bit_cast<VectorType>(0x5f3759dfu - (bits >> 1));
631 return guess * (1.5f - 0.5f * vec_ * guess * guess);
632#else
633 return simd::reciprocal_sqrt_estimate(vec_);
634#endif
635 }

814 template <bool max>
815 ace argon_type NumberMinMax(argon_type b) const {
816 using bits_type = std::conditional_t<sizeof(scalar_type) == 2, uint16_t,
817 std::conditional_t<sizeof(scalar_type) == 4, uint32_t, uint64_t>>;
818 const argon_type a{vec_};
819 argon_type r = max ? (a > b).Select(a, b) : (a < b).Select(a, b); // NaN in a: b
820 const auto a_bits = a.template As<bits_type>(), b_bits = b.template As<bits_type>();
821 const argon_type zeros = (max ? (a_bits & b_bits) : (a_bits | b_bits)).template As<scalar_type>();
822 r = (a == b).Select(zeros, r); // equal lanes: the same value, or zeros of either sign (+0 & -0 is +0)
