649 ace argon_type ReciprocalSqrtEstimate() const
650 requires lane_floating_point<scalar_type> || (std::is_same_v<scalar_type, uint32_t> && !mve_platform)
651 {
652#ifdef ARGON_PLATFORM_MVE
653 // MVE has no vrsqrte: take the bit-level initial guess, then one Newton-Raphson step, which lands within NEON's
654 // estimate precision (~1/256).
655 const auto bits = std::bit_cast<Bool_t<VectorType>>(vec_);
656 const VectorType guess = std::bit_cast<VectorType>(0x5f3759dfu - (bits >> 1));
657 return guess * (1.5f - 0.5f * vec_ * guess * guess);
658#else
659 return simd::reciprocal_sqrt_estimate(vec_);
660#endif
661 }

861 template <bool max>
862 ace argon_type NumberMinMax(argon_type b) const {
863 using bits_type = std::conditional_t<sizeof(scalar_type) == 2, uint16_t,
864 std::conditional_t<sizeof(scalar_type) == 4, uint32_t, uint64_t>>;
865 const argon_type a{vec_};

867 const auto a_bits = a.template As<bits_type>(), b_bits = b.template As<bits_type>();
868 const argon_type zeros = (max ? (a_bits & b_bits) : (a_bits | b_bits)).template As<scalar_type>();
869 r = (a == b).Select(zeros, r); // equal lanes: the same value, or zeros of either sign (+0 & -0 is +0)
870 return (b == b).Select(r, a); // NaN in b: a (NaN only if a is too)
