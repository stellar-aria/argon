135[[gnu::always_inline]] inline std::array<Argon<ScalarType>, Stride> load_first_n_interleaved(const ScalarType* ptr,
136 size_t count) {
137 using half_type = ArgonHalf<ScalarType>;
138 constexpr size_t half = half_type::lanes;
139 half_multi_type<Stride, ScalarType> zero;
140 for (size_t channel = 0; channel < Stride; ++channel)
141 zero.val[channel] = half_type{ScalarType{0}}.vec();

146 for (size_t channel = 0; channel < Stride; ++channel)

183template <typename N, typename S>
184[[gnu::always_inline]] inline Argon<S> load_first_n_widen(const N* ptr, size_t count) {
185 constexpr size_t lanes = Argon<S>::lanes;
186 if constexpr (sizeof(S) == 2 * sizeof(N)) {
187 // the narrow lanes fill a doubleword: load them there one by one, then widen with vmovl as a whole load would
188 const auto narrow = load_lanes<0>(ArgonHalf<N>{N{0}}, ptr, 0, count);
189 const auto wide = neon::move_long(narrow.vec());
190 return Argon<std::conditional_t<std::is_signed_v<N>, std::make_signed_t<S>, std::make_unsigned_t<S>>>{wide}
191 .template As<S>();
192 } else {
193 std::array<S, lanes> out{};
194 for (size_t i = 0; i < lanes; ++i) {
195 if (i < count)
196 out[i] = static_cast<S>(ptr[i]);
197 }
198 return Argon<S>::Load(out.data());
199 }
200}
