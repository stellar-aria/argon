252 // ── Bottom / top widening and narrowing ─────────────────────────────────────────────────────────────────────
253 // Helium widens and narrows by lane parity rather than by vector half: the "bottom" lanes are the even ones
254 // (0, 2, 4, ...) and the "top" lanes the odd ones. Each pair of narrow lanes shares the bits of one wide lane, so
255 // a narrow vector splits into WidenBottom() and WidenTop(), and NarrowBottom()/NarrowTop() write a wide vector
256 // back into those lanes: hi.NarrowTop(lo.NarrowBottom(dest)) reassembles what WidenBottom/WidenTop took apart.
257 // MVE has an instruction for each; NEON views the narrow vector as the wide type, where the bottom lane is the
258 // low half of each wide lane, and widens with shifts and narrows with a bit select.
259

269#endif

417 // ── Dot products ──────────────────────────────────────────────────────────────────────────────────────────
418 // Multiply-accumulate across the vector into a scalar, Helium's vmladav family: FIR filters, convolution and
419 // complex dot products in one instruction. The forms besides Plain treat lanes 2i and 2i+1 as a pair (the real
420 // and imaginary parts of a complex number) and, like the instructions, are signed-only.
421
422 using DotForm = argon::DotForm;
424 using dot_type = std::conditional_t<std::is_signed_v<ScalarType>, int32_t, uint32_t>;
426 using dot_long_type = std::conditional_t<std::is_signed_v<ScalarType>, int64_t, uint64_t>;
427
432 template <DotForm form = DotForm::Plain, typename S = ScalarType>
433 requires(std::is_integral_v<S> && sizeof(S) <= 4 && (form == DotForm::Plain || std::is_signed_v<S>))
434 ace dot_type DotProduct(Argon<S> b, dot_type acc = 0) const {
435#ifdef ARGON_PLATFORM_MVE
436 if constexpr (form == DotForm::Plain) {
437 return mve::multiply_add_dual_reduce_add_accumulate(acc, this->vec_, b.vec());
438 } else if constexpr (form == DotForm::Exchange) {
439 return mve::multiply_add_dual_reduce_add_accumulate_exchange_pairs(acc, this->vec_, b.vec());
440 } else if constexpr (form == DotForm::Subtract) {
441 return mve::multiply_subtract_dual_reduce_add_accumulate(acc, this->vec_, b.vec());
442 } else {
443 return mve::multiply_subtract_dual_reduce_add_accumulate_exchange_pairs(acc, this->vec_, b.vec());
444 }
445#else
446 using U = std::make_unsigned_t<dot_type>; // wrapping arithmetic
447 if constexpr (form == DotForm::Plain) {
448 // A sum doesn't care about lane order, so use NEON's own shape: multiply-long the low and high halves
449 // (smull/smull2), widen pairwise (vpaddl), and reduce the 32-bit lanes.
450 dot_type lanes_sum;
451 if constexpr (sizeof(S) == 1) {
452 auto low = this->GetLow().MultiplyLong(b.GetLow()), high = this->GetHigh().MultiplyLong(b.GetHigh());
453 lanes_sum = (Argon<dot_type>{neon::pairwise_add_long(low.vec())} +
454 Argon<dot_type>{neon::pairwise_add_long(high.vec())})
455 .ReduceAdd();
456 } else if constexpr (sizeof(S) == 2) {
457 lanes_sum = (this->GetLow().MultiplyLong(b.GetLow()) + this->GetHigh().MultiplyLong(b.GetHigh())).ReduceAdd();
458 } else {
459 lanes_sum = (*this * b).ReduceAdd();
460 }
461 return static_cast<dot_type>(static_cast<U>(acc) + static_cast<U>(lanes_sum));
462 } else {
463 U sum = static_cast<U>(acc);
464 for (auto p : DotProducts<form>(b)) sum += static_cast<U>(p);
465 return static_cast<dot_type>(sum);
466 }
467#endif
468 }

538 // ── Circular-buffer indices ───────────────────────────────────────────────────────────────────────────────
539 // Index vectors for walking a ring buffer (a delay line, a wavetable), ready to feed a gather. MVE generates each
540 // in one instruction (viwdup / vdwdup); NEON replays the same lane-by-lane rule.
541
548 template <int step = 1, typename S = ScalarType>
549 requires(std::is_unsigned_v<S> && sizeof(S) <= 4 && (step == 1 || step == 2 || step == 4 || step == 8))
550 ace static Argon<S> CircularIndices(uint32_t& offset, uint32_t size) {
551#ifdef ARGON_PLATFORM_MVE
552 // Write back through a local: GCC 16 at -O0 rejects the _wb intrinsics given any other pointer
553 // ("unrecognizable insn"), including through a wrapper.
554 uint32_t next = offset;
555 typename Argon<S>::vector_type indices;
556 if constexpr (sizeof(S) == 1) {
557 indices = viwdupq_wb_u8(&next, size, step);
558 } else if constexpr (sizeof(S) == 2) {
559 indices = viwdupq_wb_u16(&next, size, step);
560 } else {
561 indices = viwdupq_wb_u32(&next, size, step);
562 }
563 offset = next;
564 return indices;
565#else
566 std::array<S, lanes> out{};
567 for (auto& lane : out) {
568 lane = static_cast<S>(offset);
569 offset += step;
570 if (offset == size) offset = 0;
571 }
572 return Argon<S>::Load(out.data());
573#endif
574 }

583 ace static Argon<S> CircularIndicesDown(uint32_t& offset, uint32_t size) {
584#ifdef ARGON_PLATFORM_MVE
585 uint32_t next = offset; // see CircularIndices
586 typename Argon<S>::vector_type indices;

591 } else {
592 indices = vdwdupq_wb_u32(&next, size, step);
593 }
594 offset = next;
595 return indices;
596#else
597 std::array<S, lanes> out{};
598 for (auto& lane : out) {
599 lane = static_cast<S>(offset);
600 if (offset == 0) offset = size;
601 offset -= step;
602 }
603 return Argon<S>::Load(out.data());
604#endif

841 template <typename S = ScalarType>
842 requires(std::is_integral_v<S> && sizeof(S) <= 4)
843 ace Argon<S> BitReverse(int bits) const {
844#ifdef ARGON_PLATFORM_MVE
845 return mve::bit_reverse_shift_right(this->vec_, static_cast<int32_t>(bits));
846#else
847 using U = std::make_unsigned_t<S>;
848 constexpr int width = 8 * sizeof(S);
849 if (bits <= 0) return Argon<S>{S(0)};
850 // Reverse the bits of each byte, then the bytes of each lane.
851 Argon<uint8_t> bytes = this->template As<uint8_t>();
852#ifdef __aarch64__
853 bytes = neon::reverse_bits(bytes.vec()); // rbit
854#else
855 bytes = ((bytes >> 1) & Argon<uint8_t>{uint8_t{0x55}}) | ((bytes & Argon<uint8_t>{uint8_t{0x55}}) << 1);
856 bytes = ((bytes >> 2) & Argon<uint8_t>{uint8_t{0x33}}) | ((bytes & Argon<uint8_t>{uint8_t{0x33}}) << 2);
857 bytes = (bytes >> 4) | (bytes << 4);
858#endif
859 if constexpr (sizeof(S) == 2) {
860 bytes = bytes.Reverse16bit();
861 } else if constexpr (sizeof(S) == 4) {
862 bytes = bytes.Reverse32bit();
863 }
864 const auto reversed = bytes.template As<U>();
865 return (bits >= width ? reversed : reversed >> (width - bits)).template As<S>(); // logical shift
866#endif
867 }

869 // ── Carry chains ──────────────────────────────────────────────────────────────────────────────────────────
870 // The vector as one 128-bit number, lane 0 least significant, for multi-word arithmetic and bitstreams. MVE has
871 // instructions for these (vadc, vsbc, vshlc); NEON works through the 32-bit words one at a time.
872
875 template <typename S = ScalarType>
876 requires(std::is_integral_v<S> && sizeof(S) == 4)
877 ace Argon<S> AddWithCarry(Argon<S> b, uint32_t& carry) const {
878#ifdef ARGON_PLATFORM_MVE
879 unsigned c = carry & 1; // through a local: see CircularIndices
880 Argon<S> out;
881 if constexpr (std::is_signed_v<S>) {
882 out = vadcq_s32(this->vec_, b.vec(), &c);
883 } else {
884 out = vadcq_u32(this->vec_, b.vec(), &c);
885 }
886 carry = c;
887 return out;
888#else
889 const auto x = this->template As<uint32_t>().to_array(), y = b.template As<uint32_t>().to_array();
890 std::array<uint32_t, 4> sum{};
891 uint64_t c = carry & 1;
892 for (size_t i = 0; i < 4; ++i) {
893 c += uint64_t{x[i]} + y[i];
894 sum[i] = static_cast<uint32_t>(c);
895 c >>= 32;
896 }
897 carry = static_cast<uint32_t>(c);
898 return Argon<uint32_t>::Load(sum.data()).template As<S>();
899#endif
900 }

904 template <typename S = ScalarType>
905 requires(std::is_integral_v<S> && sizeof(S) == 4)
906 ace Argon<S> SubtractWithBorrow(Argon<S> b, uint32_t& borrow) const {
907#ifdef ARGON_PLATFORM_MVE
908 unsigned c = 1 - (borrow & 1);
909 Argon<S> out;
910 if constexpr (std::is_signed_v<S>) {
911 out = vsbcq_s32(this->vec_, b.vec(), &c);
912 } else {
913 out = vsbcq_u32(this->vec_, b.vec(), &c);
914 }
915 borrow = 1 - c;
916 return out;
917#else
918 const auto x = this->template As<uint32_t>().to_array(), y = b.template As<uint32_t>().to_array();

923 difference[i] = static_cast<uint32_t>(uint64_t{x[i]} - subtrahend);

928#endif

1277 template <DotForm form, typename S>
1278 ace std::array<std::conditional_t<std::is_signed_v<S>, int64_t, uint64_t>, Argon<S>::lanes> DotProducts(
1279 Argon<S> b) const {
1280 using T = std::conditional_t<std::is_signed_v<S>, int64_t, uint64_t>;
1281 constexpr bool exchange = form == DotForm::Exchange || form == DotForm::SubtractExchange;
1282 const auto x = this->to_array();
1283 const auto y = b.to_array();
1284 std::array<T, Argon<S>::lanes> out{};
1285 for (size_t i = 0; i < out.size(); ++i) {
1286 const size_t j = exchange ? (i ^ 1) : i; // the other lane of the pair
1287 T p = static_cast<T>(x[i]) * static_cast<T>(y[j]);
1288 // Subtract negates the odd lanes' products; SubtractExchange the even lanes' (a[2i+1]·b[2i] − a[2i]·b[2i+1]).
1289 const bool negate = form == DotForm::Subtract ? (i & 1) : (form == DotForm::SubtractExchange && !(i & 1));
1290 if (negate) p = static_cast<T>(T{0} - p);
1291 out[i] = p;
1292 }
1293 return out;
1294 }
